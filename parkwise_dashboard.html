Equivalent C Logic (Arrays, Structs, Loops & Conditions)

#define TOTAL_SLOTS 150

typedef struct {
    int id;
    int level;
    int occupied;
    char plate[32];
    char vehicle_type[24];
    char entry_time[32];
} ParkingSlot;

ParkingSlot slots[TOTAL_SLOTS];

int assign_slot(const char *plate, const char *type, int pref_level) {
    for (int i = 0; i < TOTAL_SLOTS; i++) {
        if (!slots[i].occupied && (pref_level == 0 || slots[i].level == pref_level)) {
            slots[i].occupied = 1;
            strcpy(slots[i].plate, plate);
            strcpy(slots[i].vehicle_type, type);
            return slots[i].id;
        }
    }
    return -1;
}
